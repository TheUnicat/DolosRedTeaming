import os, subprocess

def thumbnail(user_filename):
    out = os.path.join('/var/www/thumbs', user_filename + '.png')
    subprocess.run(f"convert {user_filename} -resize 128x128 {out}", shell=True)
    os.chmod(out, 0o777)
    return out
